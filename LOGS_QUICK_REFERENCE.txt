🌐 VISOR WEB
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
URL:    http://localhost:8000/logs
Inicio: php artisan serve

Características:
  ✓ Interfaz gráfica moderna (tema oscuro)
  ✓ Selector de archivos de log
  ✓ Control de líneas a mostrar
  ✓ Búsqueda en tiempo real
  ✓ Auto-actualización cada 5s
  ✓ Descarga de archivos
  ✓ Limpieza de logs
  ✓ Resaltado de colores por nivel

💻 SCRIPT BASH (./logs.sh)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Comando              Descripción
────────────────────────────────────────────────────────────────
./logs.sh            Ver últimas 50 líneas
./logs.sh -v 100     Ver últimas 100 líneas
./logs.sh -t         Seguir en tiempo real
./logs.sh -s "term"  Buscar término específico
./logs.sh -e         Mostrar solo errores
./logs.sh -w         Mostrar solo warnings
./logs.sh -c         Limpiar archivo de log
./logs.sh -p         Usar Laravel Pail
./logs.sh -h         Ver ayuda completa

Ejemplos útiles:
  ./logs.sh -v 200 | grep "error"    # Ver errores
  ./logs.sh -s "Exception"           # Buscar excepciones
  ./logs.sh -f laravel.log -t        # Tail de archivo específico

⚙️  COMANDO ARTISAN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
php artisan logs:view                       # Ver 50 líneas
php artisan logs:view --lines=100           # Ver 100 líneas
php artisan logs:view --tail                # Seguir en tiempo real
php artisan logs:view --clear               # Limpiar log
php artisan logs:view --file=custom.log     # Archivo específico

🚀 LARAVEL PAIL
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Características Pail:
  ✓ Colorización avanzada
  ✓ Filtrado en tiempo real
  ✓ Formato mejorado
  ✓ Mejor rendimiento

🎯 CASOS DE USO COMUNES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Desarrollo:
  • Terminal 1: php artisan serve
  • Terminal 2: ./logs.sh -t
  • Navegador: http://localhost:8000/logs (auto-refresh ON)

Debugging:
  • ./logs.sh -e              # Ver todos los errores
  • ./logs.sh -s "Exception"  # Buscar excepciones
  • Visor web + búsqueda      # Interfaz visual

Producción:
  • ssh a servidor
  • ./logs.sh -v 500          # Ver últimos logs
  • ./logs.sh -s "error"      # Buscar problemas
  • ./logs.sh -c              # Limpiar periódicamente

🔒 SEGURIDAD
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Desarrollo:   Acceso libre (localhost)
Producción:   Activar middleware LogViewerAccess

En routes/web.php, cambiar:
  Route::prefix('logs')->group(function () {
a:
  Route::prefix('logs')->middleware(['auth'])->group(function () {

Configurar IPs permitidas en .env:
  LOG_VIEWER_ALLOWED_IPS=127.0.0.1,192.168.1.100

🆘 PROBLEMAS COMUNES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Problema: Archivo no encontrado
Solución: Verifica que storage/logs/laravel.log existe

Problema: Sin permisos
Solución: chmod -R 755 storage/logs

Problema: Visor web no carga
Solución: Verifica php artisan serve está corriendo

Problema: Script no ejecutable
Solución: chmod +x logs.sh
